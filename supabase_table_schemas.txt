create table public."Appointment" (
  "appointmentID" uuid not null default gen_random_uuid (),
  "patientID" uuid not null,
  "doctorID" uuid not null,
  appointment_date date not null,
  time_slot time without time zone not null,
  status character varying not null default 'pending'::character varying,
  created_at timestamp without time zone null default now(),
  zoom_link text null,
  rating integer null,
  constraint Appointment_pkey primary key ("appointmentID"),
  constraint Appointment_doctorID_appointment_date_time_slot_key unique ("doctorID", appointment_date, time_slot),
  constraint Appointment_doctorID_fkey foreign KEY ("doctorID") references "Doctor" ("doctorID") on delete CASCADE,
  constraint Appointment_patientID_fkey foreign KEY ("patientID") references "Patient" ("patientID") on delete CASCADE
) TABLESPACE pg_default;


create table public."Doctor" (
  "doctorID" uuid not null default gen_random_uuid (),
  name character varying not null,
  email character varying not null,
  date_created date not null,
  specialty character varying not null,
  pfp character varying null,
  contact_num character varying null,
  constraint doctor_pkey primary key ("doctorID"),
  constraint doctor_contact_num_key unique (contact_num),
  constraint doctor_email_key unique (email),
  constraint doctor_pfp_key unique (pfp)
) TABLESPACE pg_default;

create table public."Medical History" (
  id bigint generated by default as identity not null,
  medication character varying not null,
  allergies character varying null,
  family_history character varying null,
  consent_data character varying null
) TABLESPACE pg_default;

create table public."Patient" (
  "patientID" uuid not null,
  name character varying not null,
  email character varying not null,
  contact_num character varying null,
  date_created timestamp without time zone null default now(),
  pfp character varying null,
  age smallint null,
  sex character varying null,
  constraint Patient_pkey primary key ("patientID"),
  constraint Patient_email_key unique (email),
  constraint Patient_patientID_fkey foreign KEY ("patientID") references auth.users (id) on delete CASCADE
) TABLESPACE pg_default;

create table public."Schedule" (
  "scheduleID" uuid not null default gen_random_uuid (),
  "doctorID" uuid not null,
  available_date date not null,
  time_slot time without time zone not null,
  is_available boolean not null default true,
  created_at timestamp without time zone null default now(),
  constraint Schedule_pkey primary key ("scheduleID"),
  constraint Schedule_doctorID_available_date_time_slot_key unique ("doctorID", available_date, time_slot),
  constraint Schedule_doctorID_fkey foreign KEY ("doctorID") references "Doctor" ("doctorID") on update CASCADE on delete CASCADE
) TABLESPACE pg_default;
